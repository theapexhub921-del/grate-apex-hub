    // Starred questions (new app): one document per question a learner starred. Owner only.
    match /progress/{uid}/starred/{sid} {
      allow read, delete: if request.auth != null && request.auth.uid == uid;
      allow create: if request.auth != null && request.auth.uid == uid
        && sid.size() <= 300
        && request.resource.data.keys().hasOnly(['questionId', 'starredAt'])
        && request.resource.data.questionId is string && request.resource.data.questionId.size() <= 200
        && request.resource.data.starredAt == request.time;
      allow update: if false;
    }
